```python
from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable

@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float

def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode('utf-8'):
        h = ((h ^ b) * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h

def run_benchmark(fn: Callable[[], None], warmup: int = 2, runs: int = 7) -> BenchStats:
    for _ in range(warmup): fn()
    samples = []
    for _ in range(runs):
        t0 = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns()-t0)/1e6)
    samples.sort()
    return BenchStats(samples[0], samples[runs//2], samples[(3*runs)//4]-samples[runs//4])

def emit_verify_tag(phase: str, seed: int, m1: float, m2: float) -> str:
    hw = mp.cpu_count()
    sig = fnv1a_hash(f'{phase}|{seed}|{hw}|{m1:.4f}|{m2:.4f}')
    return f'VERIFY_HASH[{phase}]: {sig:016X} (P_log={hw})'
```

```python
    elif mode=='cyclic':

    else:
        while True:
            with lock:
                start=counter.value;counter.value+=chunk
            if start>=n:break
            for i in range(start,min(start+chunk,n)):acc+=work(i,scale)
    times[slot]=time.perf_counter()-t

def trial(p,n,scale,mode,chunk):
    counter=mp.Value('i',0,lock=False);lock=mp.Lock();times=mp.Array('d',p,lock=False)
    start=time.perf_counter();ps=[]
    for j in range(p):
        proc=mp.Process(target=worker,args=(j,p,n,scale,mode,chunk,counter,lock,times));proc.start();ps.append(proc)
    for proc in ps:
        proc.join()
        if proc.exitcode:raise RuntimeError('scheduling worker failed')
    wall=time.perf_counter()-start
    vals=list(times)
    imbalance=max(vals)/(sum(vals)/p) if sum(vals)>0 else 0
    return wall,imbalance,vals
```
